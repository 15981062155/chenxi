JavaScript 条件与循环

姓名：陈曦 · 第 15 章 · 程序结构演示

一、条件结构

1. if / else if / else
// 根据分数评级 var cx_score = 85; var cx_grade; if (cx_score >= 90) { cx_grade = "A 优秀"; } else if (cx_score >= 75) { cx_grade = "B 良好"; } else if (cx_score >= 60) { cx_grade = "C 及格"; } else { cx_grade = "D 不及格"; } console.log(cx_grade);
2. switch 多分支
// 根据星期几输出安排 var cx_day = 3; // 0=周日 1=周一 ... var cx_plan; switch (cx_day) { case 1: case 2: case 3: cx_plan = "上课学习"; break; case 4: case 5: cx_plan = "做作业 + 运动"; break; default: cx_plan = "休息娱乐"; }
3. 三元运算符（简写的 if/else）
var cx_age = 20; var cx_status = (cx_age >= 18) ? "成年人" : "未成年";

二、循环结构

1. for 循环（计数循环）

输出 1~10，遇到 7 的倍数跳过（continue），遇到 >8 停止（break）

for (var i = 1; i <= 10; i++) { if (i % 7 === 0) continue; if (i > 8) break; console.log(i); }
2. while 循环（条件循环）
// 计算 1~100 的和 var cx_sum = 0; var cx_i = 1; while (cx_i <= 100) { cx_sum += cx_i; cx_i++; }
3. do...while（至少执行一次）
var cx_n = 0; do { cx_n++; } while (cx_n < 5);
4. for...of（遍历数组 / 可迭代对象）
var cx_fruits = ["苹果", "香蕉", "橙子", "葡萄"]; for (var cx_fruit of cx_fruits) { console.log(cx_fruit); }
5. for...in（遍历对象键名）
var cx_student = { name: "陈曦", age: 20, school: "阳光大学" }; for (var cx_key in cx_student) { console.log(cx_key + " = " + cx_student[cx_key]); }
💡 小提示：break 立即跳出整个循环；continue 跳过当前这一次，继续下一轮。
